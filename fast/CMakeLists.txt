cmake_minimum_required(VERSION 3.16)
project(SplitAlignerFast VERSION 0.2.0 LANGUAGES C CXX)
set(SAF_BUILD_COMMIT "91860d941c6b591b078fc99b02bdf7933fdba6c6" CACHE STRING "Source revision recorded in CLI manifests")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SAF_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SAF_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(saf_shared SHARED src/fast_core.cpp src/fast_c_api.cpp)
set_target_properties(saf_shared PROPERTIES OUTPUT_NAME splitaligner_fast)
target_include_directories(saf_shared PUBLIC include)
add_library(saf_static STATIC src/fast_core.cpp src/fast_c_api.cpp)
set_target_properties(saf_static PROPERTIES OUTPUT_NAME splitaligner_fast)
target_include_directories(saf_static PUBLIC include)
add_library(saf_collision STATIC src/fast_core.cpp src/fast_c_api.cpp)
target_compile_definitions(saf_collision PRIVATE SAF_TEST_FORCE_HASH_COLLISION)
target_include_directories(saf_collision PUBLIC include)
foreach(t saf_shared saf_static saf_collision)
 target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
add_executable(native_tests tests/native_tests.cpp)
target_link_libraries(native_tests PRIVATE saf_static Threads::Threads)
add_executable(collision_tests tests/native_tests.cpp)
target_link_libraries(collision_tests PRIVATE saf_collision Threads::Threads)
add_executable(c_smoke tests/native_c_smoke.c)
target_link_libraries(c_smoke PRIVATE saf_shared)
set_target_properties(c_smoke PROPERTIES LINKER_LANGUAGE CXX)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/cli.cpp")
 add_executable(saf src/cli.cpp)
 target_link_libraries(saf PRIVATE saf_static OpenSSL::Crypto)
 target_compile_definitions(saf PRIVATE SAF_BUILD_COMMIT="${SAF_BUILD_COMMIT}")
endif()
enable_testing()
add_test(NAME native COMMAND native_tests)
add_test(NAME collision COMMAND collision_tests)
add_test(NAME c_abi COMMAND c_smoke)

add_executable(memory_benchmark tests/memory_benchmark.cpp)
target_link_libraries(memory_benchmark PRIVATE saf_shared)

include(GNUInstallDirs)
install(TARGETS saf saf_shared saf_static
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES adapters/splitaligner_fast.py DESTINATION ${CMAKE_INSTALL_DATADIR}/SplitAlignerFast/python)
install(FILES LICENSE README.md CITATION.cff RELEASE_NOTES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/SplitAlignerFast)
install(DIRECTORY docs DESTINATION ${CMAKE_INSTALL_DATADIR}/SplitAlignerFast)
install(DIRECTORY examples DESTINATION ${CMAKE_INSTALL_DATADIR}/SplitAlignerFast
        PATTERN "quickstart.sh" EXCLUDE)
install(PROGRAMS examples/quickstart.sh DESTINATION ${CMAKE_INSTALL_DATADIR}/SplitAlignerFast/examples)


find_package(Python3 3.8 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME python_binding
           COMMAND ${CMAKE_COMMAND} -E env SAF_LIBRARY=$<TARGET_FILE:saf_shared>
                   PYTHONDONTWRITEBYTECODE=1 ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_binding.py)
  add_test(NAME cli_layered_groups
           COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_layered_groups.py $<TARGET_FILE:saf>)
  add_test(NAME cli_quartet
           COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_quartet_example.py $<TARGET_FILE:saf>)
endif()
